<template>
  <div class="step-plans animate-fade-in max-w-7xl mx-auto p-4 bg-sky-100">
    <h2 class="text-2xl font-black text-[#00aae4] italic mb-4">AFÍLIATE</h2>
    <p class="text-gray-700 font-semibold text-md mb-6 italic">Genera aquí tu cotización para {{
      store.datosMascota.nombreMascota }}</p>

    <div class="grid grid-cols-1 md:grid-cols-3 lg:grid-cols-3 gap-6 items-stretch">

      <div v-for="(detalles, nombrePlan) in obtenerPlanes" :key="nombrePlan"
        class="flex flex-col border border-gray-200 rounded-[2.5rem] overflow-hidden shadow-sm hover:shadow-md transition-all relative bg-white">

        <div class="p-6 max-w-7xl flex-grow">
          <div class="flex items-center gap-2 mb-4">
            <!-- <div
              class="bg-[#00aae4] text-white rounded-full w-8 h-8 flex items-center justify-center font-black text-2xl shrink-0 italic">
              1
            </div> -->

            <div class="bg-white p-2">
              <img :src="mapImagenes[nombrePlan.toUpperCase()]" alt="imagen del plan" class="w-80 h-80 mb-4">
            </div>
          </div>

          <p class="text-[12px] font-bold text-gray-400 mb-4 leading-tight uppercase italic text-pretty">
            <!-- Cobertura completa de accidentes y enfermedades -->
            {{ mapDescPlan[nombrePlan.toUpperCase()] }}
          </p>

          <ul class="space-y-4 px-2 p-2">
            <li v-for='(beneficio, i) in obtenerBeneficiosPorPlan(nombrePlan)' :key="i"
              class="flex items-start gap-2 text-sm text-gray-600 font-bold leading-tight uppercase">
              <span :class=mapsColorPaws[nombrePlan.toUpperCase()]> <i class="fa-solid fa-paw"></i> </span>
              <!-- <span>{{ beneficio[plan] }}</span> -->
              <span>{{ beneficio }}</span>
            </li>
          </ul>
        </div>

        <div class="bg-gray-50 p-6 border-y border-dashed border-gray-300">

          <div class="flex items-center gap-4 mb-4">

            <span class="text-[#ff6600] font-black text-xs uppercase italic tracking-tighter">
              Cobertura Medicina Preventiva
            </span>
          </div>

          <div class="flex justify-end gap-6 text-[10px] font-black text-[#ff6600] mb-2 px-2 uppercase italic">
            <span>Premium</span>
            <span v-show="modoCobertura[nombrePlan] === 'Basico'">Básico</span>
          </div>

          <div class="space-y-2">
            <div v-for="item in itemsPreventiva" :key="item"
              class="flex items-center justify-between text-[10px] max-w-7xl bg-white border border-gray-100 px-2 py-2.5 rounded-md shadow-sm">

              <div class="flex items-center gap-1 font-bold text-gray-600 italic uppercase ">
                <span class="text-[#ff6600] text-[8px]"><i class="fa-solid fa-paw"></i></span>{{ item }}
              </div>

              <div class="flex gap-10 px-2">

                <div class="rounded-full bg-[#ff6600] w-5 h-5 flex items-center justify-center">
                  <span class="text-white font-black text-sm">✓</span>
                </div>

                <div v-show="modoCobertura[nombrePlan] === 'Basico'"
                  class="rounded-full w-5 h-5 flex items-center justify-center"
                  :class="planTieneBasico(nombrePlan, item) ? 'bg-[#ff6600]' : 'border-2 border-gray-300'">
                  <span v-if="planTieneBasico(nombrePlan, item)" class="text-white font-black text-xs">✓</span>
                  <span v-else>○</span>
                </div>

              </div>
            </div>

            <div class="flex flex-row gap-4 items-end justify-end">
              <label>
                Premium
                <input type="radio" :name="'modo-' + nombrePlan" value="Premium" v-model="modoCobertura[nombrePlan]">
              </label>
              <label>
                Basico
                <input type="radio" :name="'modo-' + nombrePlan" value="Basico" v-model="modoCobertura[nombrePlan]">
              </label>
            </div>

          </div>
        </div>

        <div class="p-6 bg-[#f0f9ff] flex flex-col items-center">
          <div class="flex items-center gap-3 mb-6 w-full">
            <!-- <div
            class="bg-[#00aae4] text-white rounded-full w-8 h-8 flex items-center justify-center font-black text-lg shrink-0 italic">
            3
          </div> -->

            <div class="flex flex-col gap-1 w-full font-bold text-[11px] text-[#003366] italic uppercase">
              <label v-for="p in [70, 80, 90]" :key="p" class="flex items-center gap-2 cursor-pointer">
                <input type="radio" :name="' cob-' + nombrePlan" :value="p" v-model="store.datosMascota.cobertura"
                  class="accent-[#800000] w-3 h-3">
                Cobertura {{ p }}%
              </label>
            </div>
          </div>

          <div class="text-center mb-6">
            <div class="text-3xl font-black text-[#003366] italic tracking-tighter leading-none">
              ${{ formatPrecio(detalles) }}/mes
            </div>
          </div>

          <button @click="seleccionarPlan(nombrePlan)"
            :class="store.datosMascota.planSeleccionado === nombrePlan ? 'bg-[#00aae4] text-white shadow-inner' : 'bg-white text-[#00aae4] border-sky-300'"
            class="w-full py-2.5 font-black rounded-xl border-2 transition-all text-sm uppercase italic tracking-widest">
            {{ store.datosMascota.planSeleccionado === nombrePlan ? 'Seleccionado' : 'Seleccionar' }}
          </button>
        </div>
      </div>
    </div>

    <div class="mt-8 flex justify-between items-center">
      <button @click="$emit('prev')" class="text-[#00aae4] font-black text-sm uppercase italic hover:underline">
        ← Volver atrás
      </button>
      <button @click="$emit('next')" :disabled="!store.datosMascota.planSeleccionado"
        class="px-10 py-3 bg-[#00aae4] text-white font-black rounded-xl uppercase italic disabled:bg-gray-300 transition-all">
        Continuar
      </button>
    </div>
  </div>

<script setup>
import { computed, reactive } from 'vue';
import { useMascotaStore } from '../stores/mascotasStore';
import { tablaPrecios } from '../utils/preciosVets';
import { imagenes } from "../utils/images"
import { descPlan } from '../utils/descPlan'
import { colorPaws } from '../utils/colorPaws'

const store = useMascotaStore();
const emit = defineEmits(['next', 'prev']);

const itemsPreventiva = ['Valoraciones preventivas', 'Vacunación', 'Desparasitación', 'Asistencia 24/7', 'Esterilización', 'Servicio funerario', 'Odontología preventiva'];

const mapImagenes = { DIAMANTE: imagenes.DIAMANTE.img, ESMERALDA: imagenes.ESMERALDA.img, SILVER: imagenes.SILVER.img }
const mapDescPlan = { DIAMANTE: descPlan.DIAMANTE.desc, ESMERALDA: descPlan.ESMERALDA.desc, SILVER: descPlan.SILVER.desc }
const mapsColorPaws = { DIAMANTE: colorPaws.DIAMANTE.color, ESMERALDA: colorPaws.ESMERALDA.color, SILVER: colorPaws.SILVER.color }

const modoCobertura = reactive({
  Diamante: 'Premium',
  Esmeralda: 'Premium',
  Silver: 'Premium'
})

// Lógica de filtrado de planes según el Store
const obtenerPlanes = computed(() => {
  const { especie, edad, peso, esDeRaza } = store.datosMascota;

  // Caso Senior
  if (especie === 'Perros' && edad === '9 o más años') {
    return tablaPrecios.Senior?.Perros || {};
  }

  // Caso Perros Normales
  if (especie === 'Perros') {
    return esDeRaza
      ? (tablaPrecios.Perros?.ConRaza?.[edad] || {})
      : (tablaPrecios.Perros?.SinRaza?.[peso]?.[edad] || {});
  }

  // Caso Gatos
  if (especie === 'Gatos') {
    const keyRaza = esDeRaza ? 'ConRaza' : 'SinRaza';
    return tablaPrecios.Gatos?.[keyRaza]?.[edad] || {};
  }

  return {};
});

const formatPrecio = (detalles) => {
  const precio = typeof detalles === 'object' ? detalles.valorMensual : detalles;
  return precio ? precio.toLocaleString('es-CO') : '0';
};

const seleccionarPlan = (nombre) => {
  store.actualizarDatos({ planSeleccionado: nombre });
};

const planTieneBasico = (plan, item) => {
  if (plan.toUpperCase() === 'SILVER') return true;
  return ['Valoraciones preventivas', 'Vacunación', 'Desparasitación'].includes(item);
};

const obtenerBeneficiosPorPlan = (plan) => {
  const beneficios = {
    'DIAMANTE': ['Servicios médicos ilimitados por enfermedad', 'Servicios médicos ilimitados por accidente', 'Atención a nivel nacional', 'Atención en cualquier veterinaria o médico en todo el país', 'Cobertura para enfermedades crónicas y degenerativas', 'Atención de condiciones congénitas y hereditarias', 'Cobertura para tratamiento contra Cáncer', 'Cirugía displasia de cadera', 'Cirugía ligamentos, tendondes y rótula'],
    'ESMERALDA': ['Servicios médicos ilimitados por enfermedad', 'Servicios médicos ilimitados por accidente', 'Atención a nivel nacional', 'Atención en cualquier veterinaria o médico en todo el país'],
    'SILVER': [
      "Servicios médicos por enfermedad, 2 eventos al año hasta $400.000/evento",
      "Servicios médicos por accidentes 2 eventos al año hasta $2.000.000/evento",
      "Consulta General 4/año",
      "Atención a nivel nacional",
      "Atención en cualquier veterinaria o médico en todo el país",
      "Medicina preventiva"
    ]
  };
  return beneficios[plan.toUpperCase()] || ['Atención veterinaria integral'];
};
</script>
